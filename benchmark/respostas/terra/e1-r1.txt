function parseValor(texto) {
  if (typeof texto !== "string") return null;

  const valor = texto.trim().replace(/^R\$\s*/, "");
  const formatoValido =
    /^(?:\d+|\d{1,3}(?:\.\d{3})+)(?:,\d{1,2})?$/.test(valor);

  if (!formatoValido) return null;

  const [parteInteira, parteDecimal = ""] = valor.split(",");
  const reais = Number(parteInteira.replace(/\./g, ""));
  const centavos = Number(parteDecimal.padEnd(2, "0"));
  const resultado = reais * 100 + centavos;

  return Number.isSafeInteger(resultado) ? resultado : null;
}

module.exports = { parseValor };
