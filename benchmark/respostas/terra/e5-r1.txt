## PLANO
Implementar contadores independentes por chave com `Map`.
Manter timestamps de chamadas aceitas e remover apenas as mais antigas que a janela.
Negar sem registrar quando o limite já foi atingido.
Testar com relógio falso para cenários básicos e de janela deslizante.

## CODIGO
```javascript
class RateLimiter {
  constructor(limite, janelaMs, agora = Date.now) {
    this.limite = limite;
    this.janelaMs = janelaMs;
    this.agora = agora;
    this.chamadasPorChave = new Map();
  }

  permitir(chave) {
    const tempoAtual = this.agora();
    const inicioDaJanela = tempoAtual - this.janelaMs;
    const chamadas = this.chamadasPorChave.get(chave) || [];

    while (chamadas.length > 0 && chamadas[0] < inicioDaJanela) {
      chamadas.shift();
    }

    if (chamadas.length >= this.limite) {
      return false;
    }

    chamadas.push(tempoAtual);
    this.chamadasPorChave.set(chave, chamadas);
    return true;
  }
}

module.exports = { RateLimiter };
```

## TESTES
```javascript
const test = require('node:test');
const assert = require('node:assert');
const { RateLimiter } = require('./ratelimiter.js');

test('permite chamadas até o limite e nega a próxima', () => {
  let tempo = 0;
  const limiter = new RateLimiter(2, 1000, () => tempo);

  assert.strictEqual(limiter.permitir('cliente'), true);
  assert.strictEqual(limiter.permitir('cliente'), true);
  assert.strictEqual(limiter.permitir('cliente'), false);
});

test('mantém contadores independentes para chaves diferentes', () => {
  let tempo = 0;
  const limiter = new RateLimiter(1, 1000, () => tempo);

  assert.strictEqual(limiter.permitir('cliente-a'), true);
  assert.strictEqual(limiter.permitir('cliente-a'), false);
  assert.strictEqual(limiter.permitir('cliente-b'), true);
});

test('expira chamadas apenas quando ficam mais velhas que a janela', () => {
  let tempo = 0;
  const limiter = new RateLimiter(1, 1000, () => tempo);

  assert.strictEqual(limiter.permitir('cliente'), true);

  tempo = 1000;
  assert.strictEqual(limiter.permitir('cliente'), false);

  tempo = 1001;
  assert.strictEqual(limiter.permitir('cliente'), true);
});

test('aplica uma janela realmente deslizante', () => {
  let tempo = 100;
  const limiter = new RateLimiter(2, 1000, () => tempo);

  assert.strictEqual(limiter.permitir('cliente'), true);

  tempo = 900;
  assert.strictEqual(limiter.permitir('cliente'), true);

  tempo = 1100;
  assert.strictEqual(limiter.permitir('cliente'), false);

  tempo = 1101;
  assert.strictEqual(limiter.permitir('cliente'), true);
});
```
