## PLANO

Guardar horários permitidos separadamente por chave.
Remover apenas chamadas mais velhas que a janela.
Negar sem registrar quando o limite for atingido.
Validar os cenários exigidos com relógio falso.

## CODIGO

```javascript
class RateLimiter {
  constructor(limite, janelaMs, agora = Date.now) {
    this.limite = limite;
    this.janelaMs = janelaMs;
    this.agora = agora;
    this.chamadasPorChave = new Map();
  }

  permitir(chave) {
    const instanteAtual = this.agora();
    const inicioDaJanela = instanteAtual - this.janelaMs;
    const chamadas = this.chamadasPorChave.get(chave) || [];

    while (chamadas.length > 0 && chamadas[0] < inicioDaJanela) {
      chamadas.shift();
    }

    if (chamadas.length >= this.limite) {
      return false;
    }

    chamadas.push(instanteAtual);
    this.chamadasPorChave.set(chave, chamadas);

    return true;
  }
}

module.exports = { RateLimiter };
```

## TESTES

```javascript
const test = require('node:test');
const assert = require('node:assert/strict');
const { RateLimiter } = require('./ratelimiter.js');

test('respeita o limite básico', () => {
  let tempo = 0;
  const limitador = new RateLimiter(2, 100, () => tempo);

  assert.equal(limitador.permitir('cliente'), true);
  assert.equal(limitador.permitir('cliente'), true);
  assert.equal(limitador.permitir('cliente'), false);
});

test('mantém contadores independentes para chaves diferentes', () => {
  let tempo = 0;
  const limitador = new RateLimiter(1, 100, () => tempo);

  assert.equal(limitador.permitir('ana'), true);
  assert.equal(limitador.permitir('ana'), false);
  assert.equal(limitador.permitir('bruno'), true);
});

test('expira apenas chamadas mais velhas que a janela', () => {
  let tempo = 0;
  const limitador = new RateLimiter(1, 100, () => tempo);

  assert.equal(limitador.permitir('cliente'), true);

  tempo = 100;
  assert.equal(limitador.permitir('cliente'), false);

  tempo = 101;
  assert.equal(limitador.permitir('cliente'), true);
});

test('aplica uma janela realmente deslizante', () => {
  let tempo = 0;
  const limitador = new RateLimiter(2, 100, () => tempo);

  assert.equal(limitador.permitir('cliente'), true);

  tempo = 60;
  assert.equal(limitador.permitir('cliente'), true);

  tempo = 100;
  assert.equal(limitador.permitir('cliente'), false);

  tempo = 101;
  assert.equal(limitador.permitir('cliente'), true);

  tempo = 160;
  assert.equal(limitador.permitir('cliente'), false);

  tempo = 161;
  assert.equal(limitador.permitir('cliente'), true);
});

test('não contabiliza chamadas negadas', () => {
  let tempo = 0;
  const limitador = new RateLimiter(1, 100, () => tempo);

  assert.equal(limitador.permitir('cliente'), true);

  tempo = 50;
  assert.equal(limitador.permitir('cliente'), false);

  tempo = 101;
  assert.equal(limitador.permitir('cliente'), true);
});
```
