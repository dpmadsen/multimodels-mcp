function diasEntre(d1, d2) {
  try {
    if (typeof d1 !== 'string' ||
        typeof d2 !== 'string') return null;
    const parseDate = (str) => {
      const match =
        str.match(/^(\d{2})\/(\d{2})\/(\d{4})$/);
      if (!match) return null;
      const day = parseInt(match[1], 10);
      const month = parseInt(match[2], 10);
      const year = parseInt(match[3], 10);
      if (month < 1 || month > 12) return null;
      const daysInMonth =
        [31,28,31,30,31,30,31,31,30,31,30,31];
      if (year % 400 === 0 ||
         (year % 4 === 0 && year % 100 !== 0)) {
        daysInMonth[1] = 29;
      }
      if (day < 1 ||
          day > daysInMonth[month - 1])
        return null;
      return { year, month, day };
    };
    const p1 = parseDate(d1);
    const p2 = parseDate(d2);
    if (!p1 || !p2) return null;
    const date1 = new Date(Date.UTC(
      p1.year, p1.month - 1, p1.day));
    const date2 = new Date(Date.UTC(
      p2.year, p2.month - 1, p2.day));
    return Math.round(Math.abs(
      date1.getTime() - date2.getTime())
      / 86400000);
  } catch (e) {
    return null;
  }
}
module.exports = { diasEntre };

function diasEntre(d1, d2) {
  if (typeof d1 !== 'string' ||
      typeof d2 !== 'string') return null;

  const parse = (s) => {
    if (!/^\d{2}\/\d{2}\/\d{4}$/.test(s))
      return null;
    const [dd, mm, yyyy] =
      s.split('/').map(Number);
    if (mm < 1 || mm > 12 || dd < 1)
      return null;
    const d = new Date(
      Date.UTC(yyyy, mm - 1, dd));
    // Truque: se a data "rolou" (31/04
    // virou 01/05), era inválida.
    if (d.getUTCFullYear() !== yyyy ||
        d.getUTCMonth() !== mm - 1 ||
        d.getUTCDate() !== dd) return null;
    return d;
  };

  const dt1 = parse(d1);
  const dt2 = parse(d2);
  if (!dt1 || !dt2) return null;

  return Math.round(
    Math.abs(dt1 - dt2) / 86400000);
}
module.exports = { diasEntre };
