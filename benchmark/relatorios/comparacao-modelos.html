function validarCPF(cpf) {
  if (typeof cpf !== "string") return false;

  const numeros = cpf.replace(/[.-]/g, "");
  if (!/^\d{11}$/.test(numeros) || /^(\d)\1{10}$/.test(numeros)) {
    return false;
  }

  const calcularDigito = (quantidade) => {
    let soma = 0;
    for (let i = 0; i < quantidade; i++) {
      soma += Number(numeros[i]) * (quantidade + 1 - i);
    }
    const resto = (soma * 10) % 11;
    return resto === 10 ? 0 : resto;
  };

  return calcularDigito(9) === Number(numeros[9]) &&
         calcularDigito(10) === Number(numeros[10]);
}

module.exports = { validarCPF };

  if (/^(\d)\1{10}$/.test(digits)) return false;

  let sum = 0;
  for (let i = 0; i < 9; i++) {
    sum += parseInt(digits[i]) * (10 - i);

  let firstDigit = (sum * 10) % 11;
  if (firstDigit === 10) firstDigit = 0;

  sum = 0;
  for (let i = 0; i < 10; i++) {
    sum += parseInt(digits[i]) * (11 - i);

  let secondDigit = (sum * 10) % 11;

  return true;
}

function validarCPF(cpf) {
  if (!cpf || typeof cpf !== 'string' || cpf.length < 11 || cpf.length > 14) return false;
  const clean = cpf.replace(/[^\d]/g, '');
  if (clean.length !== 11) return false;
  if (clean === '11111111111') return false;
  const peso1 = [10, 9, 8, 7, 6, 5, 4, 3, 2];
  const peso2 = [11, 10, 9, 8, 7, 6, 5, 4, 3, 2];
  const dig1 = clean.slice(0, 9);
  const dig2 = clean.slice(9, 11);
  const dig1Sum = dig1.split('').reduce((acc, digit) =>
    acc + parseInt(digit) * peso1[peso1.length - 1 - (acc.length || 0)], 0);
  const dig1Verif = 11 - (dig1Sum % 11);
  const dig2Sum = dig1Sum + parseInt(dig2[0]) * 2;
  const dig2Verif = 11 - (dig2Sum % 11);
  return dig1Verif === parseInt(dig2[0]) && dig2Verif === parseInt(dig2[1]);
}
